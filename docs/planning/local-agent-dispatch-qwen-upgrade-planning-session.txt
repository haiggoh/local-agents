def hotswap_get_port(model_alias: str) -> int:
    """Run local-llm-hotswap.sh to ensure model server is active and get port."""
    if not os.path.isfile(HOTSWAP_SCRIPT):
        raise FileNotFoundError(f"Hotswap script not found: {HOTSWAP_SCRIPT}")

    cmd = [HOTSWAP_SCRIPT, model_alias]
    res = subprocess.run(cmd, capture_output=True, text=True)

    if res.returncode != 0:
        print(f"[-] Hotswap failed for '{model_alias}':", file=sys.stderr)
        print(res.stderr or res.stdout, file=sys.stderr)
        sys.exit(res.returncode)

    # Search for port output (e.g., SUCCESS_PORT 8000 or listening port)
    match = re.search(r"SUCCESS_PORT\s+(\d+)", res.stdout)
    if match:
        return int(match.group(1))

    match = re.search(r"port\s+(\d+)", res.stdout)
    if match:
        return int(match.group(1))

    # Fallback to default port range scanner
    for line in res.stdout.splitlines():
        if "800" in line:
            m = re.search(r"\b(800\d)\b", line)
            if m:
                return int(m.group(1))

    return 8000


def read_files_context(file_paths: list) -> str:
    """Helper to read file contents into a string format."""
    if not file_paths:
        return ""
    
    attached_context = []
    for filepath in file_paths:
        if os.path.isfile(filepath):
            try:
                with open(filepath, "r", encoding="utf-8", errors="replace") as f:
                    content = f.read()
                attached_context.append(f"--- File: {filepath} ---\n{content}\n--- End File ---")
            except Exception as e:
                print(f"[!] Could not read file {filepath}: {e}", file=sys.stderr)
    
    if attached_context:
        return "\n\n".join(attached_context) + "\n\n"
    return ""


def run_single_prompt(port: int, model_alias: str, prompt_text: str, max_tokens: int) -> str:
    """
    Executes the librarian script for a single prompt and returns the output text.
    Returns an empty string if no output file is found.
    """
    with tempfile.TemporaryDirectory() as tmpdir:
        cmd = [
            sys.executable,
            LIBRARIAN_SCRIPT,
            "--port",
            str(port),
            "--prompt",
            prompt_text,
            "--model",
            model_alias,
            "--max-tokens",
            str(max_tokens),
            "--outdir",
            tmpdir,
        ]

        # We suppress stderr for the internal call to keep the convo clean, 
        # but you can remove stderr=subprocess.DEVNULL if you want debug info.
        res = subprocess.run(cmd, capture_output=True, text=True)

        if res.returncode != 0:
            print(f"[-] Dispatch failed with exit code {res.returncode}", file=sys.stderr)
            # In convo mode, we don't exit immediately, just warn
            return ""

        output_file = os.path.join(tmpdir, "output.txt")
        reasoning_file = os.path.join(tmpdir, "reasoning.txt")

        output_text = ""
        
        # Handle Reasoning (if exists)
        if os.path.isfile(reasoning_file):
            try:
                with open(reasoning_file, "r", encoding="utf-8") as f:
                    reasoning = f.read().strip()
                if reasoning:
                    # We include reasoning in the history for context, but maybe print it differently?
                    # For simplicity, we append it to the output text for history purposes,
                    # but visually separate it for the user.
                    output_text += f"--- Local Model Thinking ---\n{reasoning}\n----------------------------\n"
            except Exception:
                pass

        # Handle Main Output
        if os.path.isfile(output_file):
            with open(output_file, "r", encoding="utf-8") as f:
                output_text += f.read()
        else:
            output_text = "[-] No output received from local model server."

        return output_text.strip()

            print(f"User: {args.prompt}")
            if file_context:
                print(f"  (Files attached: {', '.join(args.files)})")
            
            response = run_single_prompt(port, args.model, initial_prompt, args.max_tokens)
            print(f"Assistant: {response}\n")
            
            # Add to history
            conversation_history.append({"role": "user", "content": initial_prompt})
            conversation_history.append({"role": "assistant", "content": response})

        # Start the loop
        while True:
            try:
                user_input = input("You> ")
            except (EOFError, KeyboardInterrupt):
                print("\nGoodbye.")
                break

            if not user_input.strip():
                continue

            if user_input.strip().lower() in ['exit', 'quit', 'q']:
                print("Goodbye.")
                break

            # Build the prompt for the model based on history
            # We format the history into a single text block for the model
            # Note: Depending on your model's system prompt capabilities, you might want
            # to format this differently (e.g., XML tags). For now, we use a simple text format.
            
            history_text = ""
            for turn in conversation_history:
                if turn["role"] == "user":
                    history_text += f"User: {turn['content']}\n"
                else:
                    history_text += f"Assistant: {turn['content']}\n"
            
            # Append the new input
            current_prompt = history_text + f"User: {user_input}\nAssistant:"

            # We pass the entire history + new input as the prompt
            # Note: If the history gets very long, you might hit token limits.
            # For a robust solution, you might want to truncate older history,
            # but for this script, we pass it all.
            
            response = run_single_prompt(port, args.model, current_prompt, args.max_tokens)
            print(f"Assistant: {response}\n")

            # Update history
            conversation_history.append({"role": "user", "content": user_input})
            conversation_history.append({"role": "assistant", "content": response})

    # --- SINGLE-SHOT MODE (Original Behavior) ---
    else:
        prompt_text = args.prompt
        if args.files:
            file_context = read_files_context(args.files)
            prompt_text = file_context + prompt_text

        print(f"[*] Dispatching prompt to local server (port {port})...", file=sys.stderr)
        
        output_text = run_single_prompt(port, args.model, prompt_text, args.max_tokens)
        
        if output_text:
            print(output_text)
        else:
            print("[-] No output received from local model server.", file=sys.stderr)
            sys.exit(1)


if __name__ == "__main__":
    main()
